from flask import Flask, request, send_file
from flask_cors import CORS
from ultralytics import YOLO
import cv2
import numpy as np
import io

app = Flask(__name__)
CORS(app)

print("Loading YOLO background removal model...")
yolo_model = YOLO("yolo/yolov8n-seg.pt")
person_class_id = 0

def remove_bg(image: np.ndarray) -> np.ndarray:
    white_background = np.full(image.shape, 255, dtype=np.uint8)
    person_mask = np.zeros(image.shape[:2], dtype=np.uint8)
    results = yolo_model.predict(image, stream=True, conf=0.5)

    for result in results:
        if result.masks is None:
            continue
        for mask, box in zip(result.masks.xy, result.boxes):
            if int(box.cls[0]) == person_class_id:
                points = np.int32([mask])
                cv2.fillPoly(person_mask, points, 255)

    person_mask_3ch = cv2.merge([person_mask]*3)
    foreground = cv2.bitwise_and(image, person_mask_3ch)
    background = cv2.bitwise_and(white_background, cv2.bitwise_not(person_mask_3ch))
    output_image = cv2.add(foreground, background)
    return output_image

@app.route("/")
def index():
    return "works"

@app.route("/face", methods=["POST"])
def face():
    if "image_file" not in request.files:
        return "No image uploaded", 400

    file = request.files["image_file"]
    file_bytes = np.frombuffer(file.read(), np.uint8)
    img = cv2.imdecode(file_bytes, cv2.IMREAD_COLOR)
    if img is None:
        return "Invalid image", 400

    results = yolo_model.predict(img, stream=True, conf=0.5)
    people = 0
    for result in results:
        if result.masks is None:
            continue
        for mask, box in zip(result.masks.xy, result.boxes):
            if int(box.cls[0]) == person_class_id:
                people += 1
    return {"faces": people}

@app.route("/remove_bg", methods=["POST"])
def remove_bg_route():
    if "image_file" not in request.files:
        return "No image uploaded", 400

    file = request.files["image_file"]
    file_bytes = np.frombuffer(file.read(), np.uint8)
    img = cv2.imdecode(file_bytes, cv2.IMREAD_COLOR)
    if img is None:
        return "Invalid image", 400

    output_img = remove_bg(img)

    is_success, buffer = cv2.imencode(".png", output_img)

    if not is_success:
        return "Failed to encode output image", 500

    return send_file(io.BytesIO(buffer.tobytes()), mimetype="image/png")

app.run(host="0.0.0.0", port=8081)  # run on 8081 to avoid conflict with Node
